# Requirements for redteam_mcp_testbed.py + mcp_server.py
#
# Install with pip:
#   pip install -r requirements.txt
#
# Install with uv:
#   uv pip install -r requirements.txt
#
# Python 3.10+ required (3.12+ recommended).
# On Python < 3.11, tomli is needed as a stdlib tomllib backport (included below).

# --- Core framework ---
fastapi>=0.115
uvicorn>=0.41
pydantic>=2.12
starlette>=1.0.1    # underlies fastapi/mcp; pinned explicitly rather than left transitive

# --- MCP ---
# mcp 2.x is a breaking-change major version from 1.x (FastMCP renamed to
# MCPServer, camelCase fields -> snake_case, etc.). See mcp_server.py and
# redteam_mcp_testbed.py for the v2 API usage this requires.
mcp>=2.0
fastmcp>=4.0    # fastmcp 3.x pins mcp<2 and downgrades it; 4.x is the first line compatible with mcp>=2.0

# --- LLM / agent ---
# autogen 0.14.1 needs openai>=2.30 at runtime (checked when an agent is built,
# not enforced by pip). Capped at the latest release verified end to end.
openai>=2.30.0,<=3.19.2
# AutoGen: the testbed uses the classic `import autogen` API (AssistantAgent,
# UserProxyAgent, register_function), which is maintained as AG2 and published
# on PyPI as `autogen` / `ag2` / `pyautogen`. Do NOT use autogen-agentchat>=0.4:
# that is Microsoft's rewrite, which has no `autogen` module. All of these
# packages install into the same `autogen/` directory, so install only one.
# Capped below 1.0 because AG2 1.x has not been verified against this code.
autogen>=0.14.1,<1.0
httpx>=0.28

# --- Data / plotting ---
matplotlib>=3.10

# --- Async / HTTP ---
anyio>=4.8
requests>=2.32

# --- Testing ---
pytest>=9.0

# --- Config / logging ---
python-dotenv>=1.2
structlog>=26.1
tomli>=2.2          # backport of stdlib tomllib; no-op on Python 3.11+
